📘 Полная таблица команд Arduino

Все основные команды Arduino, разделённые по категориям для удобства изучения и использования.

Категория Команда Описание Пример
⚙️ Настройка пинов pinMode(pin, mode);Настройка режима работы пина.pinMode(13, OUTPUT);
digitalWrite(pin, value);Установить HIGH или LOW.digitalWrite(13, HIGH);
digitalRead(pin);Прочитать состояние пина.digitalRead(2);
analogRead(pin);Считать аналоговый сигнал (0–1023).analogRead(A0);
analogWrite(pin, value);Выдать PWM-сигнал (0–255).analogWrite(9, 128);
tone(pin, freq);Генерация звука.tone(8, 1000);
noTone(pin);Остановить звук.noTone(8);
🔢 Переменные и типы данных intЦелое число.int x = 5;
floatДробное число.float t = 23.5;
doubleДлинное дробное (или float).double y = 3.1415;
charОдин символ.char c = 'A';
StringСтрока текста.String s = "Hello";
booleanИстина/ложь.boolean flag = true;
byteБеззнаковое 0–255.byte b = 255;
constКонстанта.const int LED = 13;
⏱️ Время и задержки delay(ms);Пауза в миллисекундах.delay(1000);
delayMicroseconds(us);Пауза в микросекундах.delayMicroseconds(500);
millis();Время с запуска (мс).millis();
micros();Время с запуска (мкс).micros();
🔁 Управляющие конструкции if / elseУсловие.if(a>10){...} else {...}
for()Цикл по счётчику.for(int i=0;i<5;i++){...}
while()Цикл с условием.while(x<10){...}
do...while()Цикл с постусловием.do {...} while(x<5);
switch...caseВыбор вариантов.switch(x){case 1:...;break;}
breakПрервать цикл.break;
continueПропустить итерацию.continue;
💬 Последовательный порт Serial.begin(9600);Инициализация порта.Serial.begin(9600);
Serial.print();Вывод без новой строки.Serial.print("Hi");
Serial.println();Вывод с новой строки.Serial.println("Hi");
Serial.read();Считать байт.Serial.read();
Serial.available();Проверить наличие данных.if(Serial.available())...
Serial.write();Отправить байт/данные.Serial.write(65);
🧮 Математические функции min(a,b);Минимум.min(3,7);
max(a,b);Максимум.max(3,7);
abs(x);Модуль числа.abs(-5);
constrain(x,a,b);Ограничение диапазона.constrain(x,0,255);
map(x,in1,in2,out1,out2);Пересчёт диапазона.map(val,0,1023,0,255);
pow(x,y);Степень.pow(2,3);
sqrt(x);Квадратный корень.sqrt(9);
sin(x);Синус (в радианах).sin(PI/2);
cos(x);Косинус.cos(0);
tan(x);Тангенс.tan(PI/4);
🧱 Основные функции void setup()Запускается один раз.void setup(){pinMode(13,OUTPUT);}
void loop()Бесконечный цикл.void loop(){digitalWrite(13,HIGH);}
⚡ Прерывания attachInterrupt(pin, ISR, mode);Подключить прерывание.attachInterrupt(0, blink, RISING);
detachInterrupt(pin);Отключить прерывание.detachInterrupt(0);
noInterrupts();Выключить прерывания.noInterrupts();
interrupts();Включить прерывания.interrupts();
📦 Побитовые операции bitRead(x,n);Чтение бита.bitRead(b,2);
bitWrite(x,n,b);Запись бита.bitWrite(val,1,1);
bitSet(x,n);Установить бит.bitSet(x,2);
bitClear(x,n);Сбросить бит.bitClear(x,1);
bit(n);Возврат 2ⁿ.bit(3);
